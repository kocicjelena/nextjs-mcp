// ============================================================================
// STEP 2: Update your .env.local
// ============================================================================

/*
Add these environment variables to your .env.local file:

FACEBOOK_APP_ID=
FACEBOOK_APP_SECRET=secret
FACEBOOK_PAGE_ACCESS_TOKEN=token

Replace "secret" and "token" with your actual Facebook credentials.

To get these credentials:

1. App ID:
   - Go to https://developers.facebook.com/
   - Select your app
   - Go to Settings > Basic
   - Copy the App ID

2. App Secret:
   - Same location as App ID in Settings > Basic
   - Click "Show" and copy it

3. Page Access Token:
   - Go to your app dashboard
   - Navigate to Tools > Graph API Explorer
   - Select your page in the dropdown
   - Select "get_page_messages" or similar permissions
   - Generate an access token
   - Or: Page Settings > Messenger > Generate new token

*/

// ============================================================================
// STEP 3: Update your lib/mcp/registry.ts
// ============================================================================

/*
Update your existing registry.ts to import and register the Facebook tools:

FROM:
---
import { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js";
import { DocEntry } from "@/types/doc-entry";
import { registerSearchTool } from "./tools/search";
import { registerFetchTool } from "./tools/fetch";

export function registerAllTools(server: McpServer, docs: DocEntry[]) {
  registerSearchTool(server, docs);
  registerFetchTool(server, docs);
}

TO:
---
import { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js";
import { DocEntry } from "@/types/doc-entry";
import { registerSearchTool } from "./tools/search";
import { registerFetchTool } from "./tools/fetch";
import { registerFacebookPostTool } from "./tools/facebookPost";           ← ADD THIS
import { registerFacebookPageInfoTool } from "./tools/facebookPageInfo";   ← ADD THIS
import { createFacebookService } from "@/lib/services/facebookService";    ← ADD THIS

export function registerAllTools(server: McpServer, docs: DocEntry[]) {
  registerSearchTool(server, docs);
  registerFetchTool(server, docs);
  
  // Facebook tools
  try {
    const fbService = createFacebookService();                            ← ADD THIS
    registerFacebookPostTool(server, fbService);                          ← ADD THIS
    registerFacebookPageInfoTool(server, fbService);                      ← ADD THIS
  } catch (error) {
    console.warn("Facebook service not configured:", error);
  }
}

Note: We wrap the Facebook tools in a try/catch because if credentials
aren't set, we want the rest of your MCP server to continue working.

*/
